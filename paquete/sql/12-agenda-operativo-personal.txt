-- Paso 1: cuentas de inicio de sesión que aún no existen
insert into auth.users (
  instance_id, id, aud, role, email, encrypted_password, email_confirmed_at,
  raw_app_meta_data, raw_user_meta_data, created_at, updated_at,
  confirmation_token, recovery_token, email_change_token_new, email_change
)
select
  '00000000-0000-0000-0000-000000000000', gen_random_uuid(), 'authenticated', 'authenticated',
  n.email, crypt('ImcaTemporal2026!', gen_salt('bf')), now(),
  '{"provider":"email","providers":["email"]}', jsonb_build_object('nombre', n.nombre),
  now(), now(), '', '', '', ''
from pl_altas_tmp n
where not exists (select 1 from auth.users u where lower(u.email) = n.email);

-- Paso 2: identidad de correo para las cuentas nuevas
insert into auth.identities (id, user_id, identity_data, provider, provider_id, last_sign_in_at, created_at, updated_at)
select gen_random_uuid(), u.id,
       jsonb_build_object('sub', u.id::text, 'email', u.email, 'email_verified', true),
       'email', u.id::text, now(), now(), now()
from auth.users u
join pl_altas_tmp n on lower(u.email) = n.email
where not exists (select 1 from auth.identities i where i.user_id = u.id);

-- Paso 3: perfil (nombre y rol de la agenda). Si ya tiene perfil, no lo cambia.
insert into perfiles (id, nombre, email, rol, activo)
select u.id, n.nombre, n.email, n.rol_perfil, true
from auth.users u
join pl_altas_tmp n on lower(u.email) = n.email
on conflict (id) do nothing;

-- Paso 4: rol y permisos de Agenda operativo
insert into pl_personal (perfil_id, rol, convocable, ve_dashboard, puede_anular, puede_descargar, activo)
select u.id, n.rol, n.convocable, n.ve_dashboard, n.puede_anular, n.puede_descargar, true
from auth.users u
join pl_altas_tmp n on lower(u.email) = n.email
on conflict (perfil_id) do update
  set rol = excluded.rol,
      convocable = excluded.convocable,
      ve_dashboard = excluded.ve_dashboard,
      puede_anular = excluded.puede_anular,
      puede_descargar = excluded.puede_descargar,
      activo = true;

-- Limpieza de la lista de trabajo
drop table if exists pl_altas_tmp;

-- Verificación: debe mostrar 17 filas
select p.nombre, p.email, pp.rol, pp.convocable, pp.ve_dashboard, pp.puede_anular, pp.puede_descargar
from pl_personal pp
join perfiles p on p.id = pp.perfil_id
order by pp.rol, p.nombre;
