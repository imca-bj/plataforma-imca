select
  p.proname as funcion,
  pg_get_functiondef(p.oid) like '%acceso_persona%'      as lee_personal,
  pg_get_functiondef(p.oid) like '%acceso_rol_atencion%' as sincroniza_atencion,
  pg_get_functiondef(p.oid) like '%pl_personal%'         as sincroniza_operativo
from pg_proc p
where p.pronamespace = 'public'::regnamespace
  and p.proname in ('mi_rol', 'acceso_rol_atencion', 'acceso_rol_operativo', 'acceso_asignar_rol')
order by p.proname;
