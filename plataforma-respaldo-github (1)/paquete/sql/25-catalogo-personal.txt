-- 1. Función para comparar nombres sin acentos y solo con las dos primeras palabras
create or replace function txt_clave(t text)
returns text language sql immutable as $$
  select array_to_string((string_to_array(
           lower(translate(coalesce(t, ''), 'ÁÉÍÓÚÜÑáéíóúüñ', 'AEIOUUNaeiouun')), ' '))[1:2], ' ');
$$;

-- 2. Catálogo
create table if not exists personal_catalogo (
  id uuid primary key default gen_random_uuid(),
  nombre text not null,
  correo text unique,
  area text not null default '',
  puesto text not null default '',
  activo boolean not null default true,
  perfil_id uuid unique references perfiles(id) on delete set null,
  creado timestamptz not null default now()
);

alter table personal_catalogo enable row level security;
drop policy if exists personal_catalogo_ver on personal_catalogo;
create policy personal_catalogo_ver on personal_catalogo for select to authenticated using (es_admin_plataforma());
revoke all on personal_catalogo from anon, authenticated;
grant select on personal_catalogo to authenticated;

-- 4. Enlazar con la cuenta de acceso (solo si el nombre coincide de forma única)
with cand as (
  select pc.id as pid, p.id as perf
  from personal_catalogo pc
  join perfiles p on txt_clave(p.nombre) = txt_clave(pc.nombre)
  where pc.perfil_id is null
),
uno as (
  select c.pid, c.perf from cand c
  where (select count(*) from cand c2 where c2.pid = c.pid) = 1
    and (select count(*) from cand c3 where c3.perf = c.perf) = 1
)
update personal_catalogo pc set perfil_id = u.perf
from uno u where pc.id = u.pid;

-- 5. Enlazar el personal de Vacaciones con el catálogo
alter table vac_personal add column if not exists personal_id uuid references personal_catalogo(id);

with cand as (
  select vp.id as vid, pc.id as pid
  from vac_personal vp
  join personal_catalogo pc on txt_clave(pc.nombre) = txt_clave(vp.nombre)
  where vp.personal_id is null
),
uno as (
  select c.vid, c.pid from cand c
  where (select count(*) from cand c2 where c2.vid = c.vid) = 1
    and (select count(*) from cand c3 where c3.pid = c.pid) = 1
)
update vac_personal vp set personal_id = u.pid
from uno u where vp.id = u.vid;
