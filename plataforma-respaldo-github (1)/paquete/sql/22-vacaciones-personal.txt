-- 22 · Vacaciones: listado de personal (carga inicial)
-- Correr en Supabase > SQL Editor, una sola vez, después del 21.
-- Cada persona inicia con 10 días de saldo. Los días tomados se capturan después.

create table if not exists vac_personal (
  id uuid primary key default gen_random_uuid(),
  nombre text not null unique,
  fecha_ingreso date,
  saldo_inicial integer not null default 10,
  activo boolean not null default true,
  creado timestamptz not null default now()
);

alter table vac_personal enable row level security;

drop policy if exists vac_personal_ver on vac_personal;
create policy vac_personal_ver on vac_personal
  for select to authenticated
  using (es_admin_plataforma());

revoke all on vac_personal from anon, authenticated;
grant select on vac_personal to authenticated;
