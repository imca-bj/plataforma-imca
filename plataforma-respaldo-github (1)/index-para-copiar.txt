<script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
<script>
  // ==== CONFIGURACIÓN ====
  const SUPABASE_URL = 'https://pighjimetgusfjbxgmlh.supabase.co';
  const SUPABASE_ANON_KEY = 'sb_publishable_MGjYCJic7OiktRYrBDhUNQ_5QiQJbFo';

  // Módulos de la plataforma y los roles que pueden entrar a cada uno.
  // Son provisionales: se ajustan cuando migremos cada módulo.
  const MODULOS = [
    { nombre: 'Agenda general',            url: '#agenda-general',   roles: ['admin','agenda','terapeuta','consulta'] },
    { nombre: 'Agenda de atención',        url: '#agenda-atencion',  roles: ['admin','agenda','terapeuta','consulta'] },
    { nombre: 'Reportes de atención',      url: '#reportes',         roles: ['admin','coordinacion','operativo','consulta'] },
    { nombre: 'Pláticas',                  url: '#platicas',         roles: ['admin','coordinacion','operativo','consulta'] },
    { nombre: 'Vacaciones y faltas',       url: '#vacaciones',       roles: ['admin','coordinacion'] },
    { nombre: 'Control de gasolina',       url: '#gasolina',         roles: ['admin','coordinacion'] }
  ];

  const sb = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

  const $ = (id) => document.getElementById(id);

  async function entrar() {
    $('error-login').textContent = '';
    const { error } = await sb.auth.signInWithPassword({
      email: $('correo').value.trim(),
      password: $('clave').value
    });
    if (error) { $('error-login').textContent = 'Correo o contraseña incorrectos.'; return; }
    await mostrarMenu();
  }

  async function mostrarMenu() {
    const { data: { user } } = await sb.auth.getUser();
    if (!user) { mostrarLogin(); return; }

    // Lee el perfil y su rol. Si no tiene perfil activo, no entra.
    const { data: perfil, error } = await sb.from('perfiles').select('nombre, rol, activo').eq('id', user.id).single();
    if (error || !perfil || !perfil.activo) {
      await sb.auth.signOut();
      $('error-login').textContent = 'Tu cuenta no tiene acceso a la plataforma. Pide que te den de alta.';
      mostrarLogin();
      return;
    }

    $('saludo').textContent = 'Hola, ' + perfil.nombre + ' (' + perfil.rol + ')';
    const lista = $('lista-modulos');
    lista.innerHTML = '';
    MODULOS.filter(m => m.roles.includes(perfil.rol)).forEach(m => {
      const a = document.createElement('a');
      a.className = 'modulo';
      a.href = m.url;
      a.textContent = m.nombre;
      lista.appendChild(a);
    });

    $('pantalla-login').classList.add('oculto');
    $('pantalla-menu').classList.remove('oculto');
    $('btn-salir').classList.remove('oculto');
  }

  function mostrarLogin() {
    $('pantalla-login').classList.remove('oculto');
    $('pantalla-menu').classList.add('oculto');
    $('btn-salir').classList.add('oculto');
  }

  async function salir() {
    await sb.auth.signOut();
    $('clave').value = '';
    mostrarLogin();
  }

  $('btn-entrar').addEventListener('click', entrar);
  $('btn-salir').addEventListener('click', salir);
  $('clave').addEventListener('keydown', (e) => { if (e.key === 'Enter') entrar(); });

  // Al abrir la página, si ya hay sesión, va directo al menú.
  mostrarMenu();
</script>
